from awsglue.utils import getResolvedOptions
from pyspark.context import SparkContext
from awsglue.context import GlueContext
from awsglue.job import Job

## @params: [JOB_NAME]

spark = glueContext.spark_session
job = Job(glueContext)
job.init(args['JOB_NAME'], args)

# --- ORDER ITEMS ---
order_items_df = spark.read.format("jdbc").options(
    **jdbc_options,
    dbtable="dbo.order_items"
).load()

order_items_df = order_items_df.withColumn("year", year(order_items_df.creation_time_utc)) \
                               .withColumn("month", month(order_items_df.creation_time_utc)) \
                               .withColumn("day", dayofmonth(order_items_df.creation_time_utc))
