CREATE DATABASE globalpartners_gold;

1. Daily Sales

CREATE EXTERNAL TABLE IF NOT EXISTS globalpartners_gold.sales_daily (
  order_date date,
  daily_revenue double,
  daily_active_customers int
)
PARTITIONED BY (
  year int,
  month int
)
STORED AS PARQUET
LOCATION 's3://globalpartners-gold-robert/sales_daily/';
